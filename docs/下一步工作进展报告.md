```python
def generate_state_machine(states: List[State]) -> str:
    # 生成Python状态机代码
    code = []
    for state in states:
        code.append(f"class {state.name}State:")
        code.append(f"    def enter(self):")
        code.append(f"        # 进入状态")
        code.append(f"    def exit(self):")
        code.append(f"        # 退出状态")
    return '\n'.join(code)
```
